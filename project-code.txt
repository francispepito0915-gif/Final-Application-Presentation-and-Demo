function formatMoney(amount) {
  return "PHP " + amount.toFixed(2);
}

function getProductPrice(product) {
  if (product.price !== undefined) {
    return product.price;
  }

  return Object.values(product.sizes)[0];
}

function getPriceText(product) {
  if (product.price !== undefined) {
    return formatMoney(product.price);
  }

  return Object.keys(product.sizes).map((size) => {
    return size + " " + formatMoney(product.sizes[size]);
  }).join(" / ");
}

function getSizeOptions(product) {
  if (!product.sizes) {
    return "";
  }

  const options = Object.keys(product.sizes).map((size) => {
    return `<option value="${size}">${size} - ${formatMoney(product.sizes[size])}</option>`;
  }).join("");

  return `
    <label class="size-label" for="size-${product.id}">Size</label>
    <select class="size-select" id="size-${product.id}" data-id="${product.id}">
      ${options}
    </select>
  `;
}

function loadProducts() {
  const savedStock = JSON.parse(localStorage.getItem(storageKey));
  const savedPrices = JSON.parse(localStorage.getItem(priceStorageKey));

  return defaultProducts.map((product) => ({
    ...product,
    price: product.price !== undefined && savedPrices && savedPrices[product.id] !== undefined ? savedPrices[product.id] : product.price,
    sizes: product.sizes && savedPrices && savedPrices[product.id] !== undefined ? savedPrices[product.id] : product.sizes,
    stock: savedStock && savedStock[product.id] !== undefined ? savedStock[product.id] : product.stock
  }));
}

function saveProducts() {
  const stockData = {};

  products.forEach((product) => {
    stockData[product.id] = product.stock;
  });

  localStorage.setItem(storageKey, JSON.stringify(stockData));
}

function savePrices() {
  const priceData = {};

  products.forEach((product) => {
    if (product.price !== undefined) {
      priceData[product.id] = product.price;
    } else {
      priceData[product.id] = product.sizes;
    }
  });

  localStorage.setItem(priceStorageKey, JSON.stringify(priceData));
}

function loadTransactionHistory() {
  const savedHistory = JSON.parse(localStorage.getItem(transactionStorageKey));

  if (!savedHistory) {
    return [];
  }

  return savedHistory;
}

function saveTransactionHistory() {
  localStorage.setItem(transactionStorageKey, JSON.stringify(transactionHistory));
}

function getStockStatus(stock) {
  if (stock === 0) {
    return "Out of Stock";
  }

  if (stock <= 5) {
    return "Low Stock";
  }

  return "Available";
}

function getStatusClass(status) {
  return status.toLowerCase().replaceAll(" ", "-");
}

function getCartQuantity(productId) {
  return cart
    .filter((item) => item.id === productId)
    .reduce((sum, item) => sum + item.quantity, 0);
}

function getTotal() {
  return cart.reduce((sum, item) => sum + item.price * item.quantity, 0);
}

function login(username, password) {
  const account = demoAccounts.find((user) => {
    return user.username === username && user.password === password;
  });

  if (!account) {
    loginMessage.textContent = "Invalid demo username or password.";
    return;
  }

  currentUser = account;
  loginMessage.textContent = "";
  usernameInput.value = "";
  passwordInput.value = "";
  loginScreen.classList.add("hidden");
  appShell.classList.remove("hidden");
  roleDisplay.textContent = "Role: " + currentUser.role;
  adminPanel.classList.toggle("hidden", currentUser.role !== "Admin");
  renderProducts();
  renderCart();
  renderAdminSections();
}

function logout() {
  currentUser = null;
  clearTransaction("");
  appShell.classList.add("hidden");
  adminPanel.classList.add("hidden");
  loginScreen.classList.remove("hidden");
}

function renderProducts() {
  const filteredProducts = selectedCategory === "All"
    ? products
    : products.filter((product) => product.category === selectedCategory);

  productGrid.innerHTML = "";

  filteredProducts.forEach((product) => {
    const quantityInCart = getCartQuantity(product.id);
    const availableStock = product.stock - quantityInCart;
    const stockStatus = getStockStatus(product.stock);
    const card = document.createElement("article");
    const firstPrice = getProductPrice(product);

    card.className = "product-card";
    card.innerHTML = `
      <div class="drink-image">
        <img src="${product.imagePath}" alt="${product.name}" onerror="this.classList.add('hidden-image'); this.nextElementSibling.classList.remove('hidden');">
        <div class="drink-icon hidden">${getDrinkSticker(product.category)}</div>
      </div>
      <div>
        <h3>${product.name}</h3>
        <div class="product-meta">
          <span>${product.category}</span>
          <span>Stock: ${product.stock}</span>
        </div>
      </div>
      <span class="stock-badge ${getStatusClass(stockStatus)}">${stockStatus}</span>
      ${getSizeOptions(product)}
      <div class="price" id="price-${product.id}">${product.sizes ? formatMoney(firstPrice) : getPriceText(product)}</div>
      <button class="add-btn" data-id="${product.id}" ${availableStock === 0 ? "disabled" : ""}>
        ${availableStock === 0 ? "Out of Stock" : "Add to Cart"}
      </button>
    `;

    productGrid.appendChild(card);
  });

  document.querySelectorAll(".add-btn").forEach((button) => {
    button.addEventListener("click", () => {
      const productId = Number(button.dataset.id);
      const sizeSelect = document.getElementById("size-" + productId);
      const selectedSize = sizeSelect ? sizeSelect.value : "Regular";

      addToCart(productId, selectedSize);
    });
  });

  document.querySelectorAll(".size-select").forEach((select) => {
    select.addEventListener("change", () => {
      const product = products.find((item) => item.id === Number(select.dataset.id));
      const priceBox = document.getElementById("price-" + product.id);

      priceBox.textContent = formatMoney(product.sizes[select.value]);
    });
  });
}

function renderAdminSections() {
  if (!currentUser || currentUser.role !== "Admin") {
    return;
  }

  const currentStockProduct = stockProductSelect.value;
  const currentPriceProduct = priceProductSelect.value;

  stockProductSelect.innerHTML = "";
  priceProductSelect.innerHTML = "";
  products.forEach((product) => {
    const stockOption = document.createElement("option");
    const priceOption = document.createElement("option");

    stockOption.value = product.id;
    stockOption.textContent = product.name + " - Current stock: " + product.stock;
    priceOption.value = product.id;
    priceOption.textContent = product.name + " - Current price: " + getPriceText(product);

    stockProductSelect.appendChild(stockOption);
    priceProductSelect.appendChild(priceOption);
  });

  stockProductSelect.value = currentStockProduct || products[0].id;
  priceProductSelect.value = currentPriceProduct || products[0].id;

  renderSalesReport();
}

function updateProductStock(productId, quantityToAdd) {
  const product = products.find((item) => item.id === productId);

  if (!quantityToAdd || quantityToAdd <= 0) {
    message.textContent = "Enter a quantity greater than zero.";
    return;
  }

  product.stock += quantityToAdd;
  saveProducts();
  message.textContent = "Stock updated.";
  stockAddInput.value = "";
  renderCart();
  renderAdminSections();
}

function updateProductPrice(productId, newPrice) {
  const product = products.find((item) => item.id === productId);

  if (!newPrice || newPrice <= 0) {
    message.textContent = "Price must be greater than zero.";
    return;
  }

  if (product.sizes) {
    Object.keys(product.sizes).forEach((size) => {
      product.sizes[size] = newPrice;
    });
  } else {
    product.price = newPrice;
  }

  savePrices();
  cart.forEach((item) => {
    if (item.id === productId) {
      item.price = newPrice;
    }
  });
  message.textContent = "Price updated.";
  newPriceInput.value = "";
  renderCart();
  renderAdminSections();
}

function renderSalesReport() {
  const totalSales = transactionHistory.reduce((sum, transaction) => {
    return sum + transaction.totalAmount;
  }, 0);
  const soldCounts = {};

  transactionHistory.forEach((transaction) => {
    transaction.items.forEach((item) => {
      soldCounts[item.name] = (soldCounts[item.name] || 0) + item.quantity;
    });
  });

  let mostSoldName = "None";
  let mostSoldQuantity = 0;

  Object.keys(soldCounts).forEach((name) => {
    if (soldCounts[name] > mostSoldQuantity) {
      mostSoldName = name;
      mostSoldQuantity = soldCounts[name];
    }
  });

  reportTransactions.textContent = transactionHistory.length;
  reportMostSold.textContent = mostSoldQuantity > 0
    ? mostSoldName + " (" + mostSoldQuantity + ")"
    : "None";
  reportSales.textContent = formatMoney(totalSales);
  renderTransactionHistory();
}

function renderTransactionHistory() {
  if (!transactionHistory.length) {
    transactionHistoryBox.innerHTML = '<p class="empty-cart">No transactions yet.</p>';
    return;
  }

  transactionHistoryBox.innerHTML = "";

  transactionHistory.slice().reverse().forEach((transaction) => {
    const card = document.createElement("div");
    const itemList = transaction.items.map((item) => {
      return `<li>${item.name} - Qty: ${item.quantity} - ${formatMoney(item.subtotal)}</li>`;
    }).join("");

    card.className = "transaction-card";
    card.innerHTML = `
      <div class="transaction-top">
        <strong>${transaction.id}</strong>
        <span>${transaction.dateTime}</span>
      </div>
      <ul>${itemList}</ul>
      <div class="transaction-money">
        <span>Total: <strong>${formatMoney(transaction.totalAmount)}</strong></span>
        <span>Cash: <strong>${formatMoney(transaction.cashPaid)}</strong></span>
        <span>Change: <strong>${formatMoney(transaction.change)}</strong></span>
      </div>
    `;

    transactionHistoryBox.appendChild(card);
  });
}

function renderCart() {
  const totalItems = cart.reduce((sum, item) => sum + item.quantity, 0);
  const total = getTotal();

  itemCount.textContent = totalItems === 1 ? "1 item" : totalItems + " items";
  totalAmount.textContent = formatMoney(total);

  if (cart.length === 0) {
    cartItems.innerHTML = '<p class="empty-cart">No drinks added yet.</p>';
  } else {
    cartItems.innerHTML = "";

    cart.forEach((item) => {
      const line = document.createElement("div");

      line.className = "cart-line";
      line.innerHTML = `
        <div>
          <h3>${item.name}</h3>
          <p>${item.size} - ${formatMoney(item.price)} x ${item.quantity}</p>
        </div>
        <div class="cart-line-actions">
          <button class="qty-btn" data-action="decrease" data-key="${item.cartKey}">-</button>
          <strong>${item.quantity}</strong>
          <button class="qty-btn" data-action="increase" data-key="${item.cartKey}">+</button>
        </div>
      `;

      cartItems.appendChild(line);
    });
  }

  document.querySelectorAll(".qty-btn").forEach((button) => {
    button.addEventListener("click", () => {
      updateQuantity(button.dataset.key, button.dataset.action);
    });
  });

  updateChange();
  renderProducts();
}

function addToCart(productId, selectedSize) {
  const product = products.find((item) => item.id === productId);
  const cartKey = productId + "-" + selectedSize;
  const cartItem = cart.find((item) => item.cartKey === cartKey);
  const quantityInCart = getCartQuantity(productId);
  const price = product.sizes ? product.sizes[selectedSize] : product.price;

  if (quantityInCart >= product.stock) {
    message.textContent = "Cannot add more than the available stock.";
    return;
  }

  if (cartItem) {
    cartItem.quantity += 1;
  } else {
    cart.push({
      cartKey: cartKey,
      id: product.id,
      name: product.name,
      category: product.category,
      size: selectedSize,
      price: price,
      quantity: 1
    });
  }

  message.textContent = "";
  renderCart();
}

function updateQuantity(cartKey, action) {
  const cartItem = cart.find((item) => item.cartKey === cartKey);

  if (!cartItem) {
    return;
  }

  const product = products.find((item) => item.id === cartItem.id);
  const quantityInCart = getCartQuantity(product.id);

  if (action === "increase") {
    if (quantityInCart < product.stock) {
      cartItem.quantity += 1;
      message.textContent = "";
    } else {
      message.textContent = "No more stock available for this drink.";
    }
  }

  if (action === "decrease") {
    cartItem.quantity -= 1;

    if (cartItem.quantity === 0) {
      cart = cart.filter((item) => item.cartKey !== cartKey);
    }

    message.textContent = "";
  }

  renderCart();
}

function updateChange() {
  const cash = Number(cashInput.value);
  const total = getTotal();
  const change = cash - total;

  changeAmount.textContent = change > 0 ? formatMoney(change) : formatMoney(0);
}

function completeTransaction() {
  const total = getTotal();
  const cash = Number(cashInput.value);
  const change = cash - total;

  if (cart.length === 0) {
    message.textContent = "Please add at least one drink.";
    return;
  }

  if (!cashInput.value || cash < total) {
    message.textContent = "Cash is not enough to complete this transaction.";
    return;
  }

  cart.forEach((cartItem) => {
    const product = products.find((item) => item.id === cartItem.id);
    product.stock -= cartItem.quantity;
  });

  const transaction = {
    id: "TXN-" + Date.now(),
    dateTime: new Date().toLocaleString(),
    items: cart.map((item) => ({
      name: item.size === "Regular" ? item.name : item.name + " (" + item.size + ")",
      quantity: item.quantity,
      price: item.price,
      subtotal: item.price * item.quantity
    })),
    totalAmount: total,
    cashPaid: cash,
    change: change
  };

  transactionHistory.push(transaction);

  saveProducts();
  saveTransactionHistory();
  cart = [];
  cashInput.value = "";
  message.textContent = "Transaction completed. Thank you!";
  renderCart();
  renderAdminSections();
}

function clearTransaction(note) {
  cart = [];
  cashInput.value = "";
  message.textContent = note;
  renderCart();
}
